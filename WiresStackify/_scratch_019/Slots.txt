	private static bool FindFreeSlotForItem(ItemData item, out Slot slot)
	{
		slot = null;
		if (item == null)
		{
			return false;
		}
		if (TryGetSavedPlayerSlot(item, out var slot2) && slot2.IsActive && slot2.ItemBelongs(item) && (slot2.IsFree || item == slot2.Item))
		{
			slot = slot2;
			return true;
		}
		int num = Array.FindIndex(slots, (Slot s) => s.IsActive && s.IsFree && s.ItemBelongs(item));
		if (num == -1)
		{
			return false;
		}
		slot = slots[num];
		return true;
	}

	private static string DescribeCaller()
	{

		{

		}

			select frame.GetMethod() into method
			where method != null
			select method.DeclaringType?.Name + "." + method.Name);
	}
