	public static bool HasAuga { get; private set; }

	private void Awake()
	{
		_instance = this;
		HasAuga = Auga.API.IsLoaded();
		if (!HasAuga && Chainloader.PluginInfos.ContainsKey("randyknapp.mods.auga"))
		{
			LogWarning("Project Auga is installed but its API did not attach to Equipment and Quick Slots; the Auga layout is off. Check the log for an APIManager error.");
		}
		new ValConfig(((BaseUnityPlugin)this).Config);
		FixQuickSlotPositionForAuga();
		LoadAssets();
		Slots.InitializeSlots();
		EpicLootCompat.Initialize();
		ItemFavoritesCompat.Initialize();
		_harmony = Harmony.CreateAndPatchAll(Assembly.GetExecutingAssembly(), "randyknapp.mods.equipmentandquickslots");
		BetterArcheryCompat.Initialize(_harmony);
	}

	private void Update()
	{
		Player localPlayer = Player.m_localPlayer;
		GamepadQuickSlots.UsePending(localPlayer);
		if ((Object)(object)localPlayer == (Object)null)
		{
			return;
		}
		Slots.Slot[] slots = Slots.slots;
		foreach (Slots.Slot slot in slots)
		{
			if (slot.IsHotkeySlot && slot.IsShortcutDownWithItem())
			{
				((Humanoid)localPlayer).UseItem((Inventory)null, slot.Item, false);
			}
		}
	}

	private void LateUpdate()
	{
		SlotValidation.Validate();
		API.DetectSlotItemChanges();
	}

	private static void LoadAssets()
	{
