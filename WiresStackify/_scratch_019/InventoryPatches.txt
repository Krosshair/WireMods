	private static class PreventStackAll
	{
		[HarmonyPatch(typeof(Inventory), "StackAll")]
		internal static class Inventory_StackAll_PreventStackingItemsFromSlots
		{
			public static bool inCall;

			[HarmonyPriority(800)]
			private static void Prefix(Inventory fromInventory)
			{
				if (inCall = fromInventory == Slots.PlayerInventory && ValConfig.PreventStackAll.Value)
				{
					RemoveItemsFromPlayerInventory();
				}
			}

			[HarmonyPriority(800)]
			private static void Finalizer()
			{
				if (inCall)
				{
					BringItemsBack();
				}
				inCall = false;
			}
		}

		[HarmonyPatch(typeof(Inventory), "Changed")]
		private static class Inventory_Changed_BringItemsBack
		{
			[HarmonyPriority(800)]
			private static void Prefix(Inventory __instance)
			{
				if (__instance == Slots.PlayerInventory && Inventory_StackAll_PreventStackingItemsFromSlots.inCall)
				{
					BringItemsBack();
				}
			}
		}

		private static readonly List<ItemData> removedItems = new List<ItemData>();

		internal static void RemoveItemsFromPlayerInventory()
		{
			for (int num = Slots.PlayerInventory.m_inventory.Count - 1; num >= 0; num--)
			{
				ItemData item = Slots.PlayerInventory.m_inventory[num];
				if (Slots.IsItemInSlot(item))
				{
					removedItems.Add(item);
					Slots.PlayerInventory.m_inventory.RemoveAt(num);
				}
			}
		}

		internal static void BringItemsBack()
		{
			if (removedItems.Count != 0)
			{
				Slots.PlayerInventory.m_inventory.AddRange(removedItems);
				removedItems.Clear();
			}
		}
	}

	private static bool PassDropItem(string source, InventoryGrid grid, Inventory fromInventory, ItemData item, Vector2i pos)
	{
		//IL_0001: Unknown result type (might be due to invalid IL or missing references)
		//IL_0006: Unknown result type (might be due to invalid IL or missing references)
		//IL_00d2: Unknown result type (might be due to invalid IL or missing references)
		//IL_00d9: Unknown result type (might be due to invalid IL or missing references)
		//IL_0089: Unknown result type (might be due to invalid IL or missing references)
		//IL_0049: Unknown result type (might be due to invalid IL or missing references)
		//IL_0050: Unknown result type (might be due to invalid IL or missing references)
		//IL_00f6: Unknown result type (might be due to invalid IL or missing references)
		if (item.m_gridPos == pos && grid.m_inventory == fromInventory)
		{
			return true;
		}
		Player localPlayer = Player.m_localPlayer;
		if (fromInventory == Slots.PlayerInventory)
		{
			Slots.Slot itemSlot = Slots.GetItemSlot(item);
			if (itemSlot != null && itemSlot.IsEquipmentSlot && ((Humanoid)localPlayer).IsItemEquiped(item) && grid.m_inventory.GetItemAt(pos.x, pos.y) != null)
			{
				EquipmentAndQuickSlots.Log($"{source}: prevented swapping equipped item {item.m_shared.m_name} out of {itemSlot}");
				return false;
			}
		}
		if (grid.m_inventory == Slots.PlayerInventory)
		{
			Slots.Slot slotInGrid = Slots.GetSlotInGrid(pos);
			if (slotInGrid != null && (!slotInGrid.ItemFits(item) || (slotInGrid.IsEquipmentSlot && !Slots.WouldFitEquipmentSlot(slotInGrid, item))))
			{
				EquipmentAndQuickSlots.Log($"{source}: prevented dropping {item.m_shared.m_name} into unfit slot {slotInGrid}");
				return false;
			}
		}
		ItemData itemAt = grid.m_inventory.GetItemAt(pos.x, pos.y);
		if (itemAt != null && itemAt != item && fromInventory == Slots.PlayerInventory)
		{
			Slots.Slot slotInGrid2 = Slots.GetSlotInGrid(item.m_gridPos);
			if (slotInGrid2 != null && !slotInGrid2.ItemFits(itemAt))
			{
				EquipmentAndQuickSlots.Log(source + ": prevented swapping " + item.m_shared.m_name + " with unfit item " + itemAt.m_shared.m_name);
				return false;
			}
		}
		return true;
	}
